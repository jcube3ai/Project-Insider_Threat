Baseline Table
Builders

Three detection models — each chosen based on the nature of the data. MAD for volatile, role-dependent behaviour. p95 for stable individual patterns. Presence detection where counting is irrelevant and the first occurrence is the signal. Using the wrong model for the wrong data is the primary cause of false positive floods in insider threat programmes.

MAD — Cohort-Based
Median Absolute Deviation. Groups users by job title and flags those who deviate significantly from their peer cohort. Robust to outliers — a partner working a massive deal doesn't corrupt the baseline for all partners. Uses only avg() and if() — no stddev() or windowcomp needed. Confirmed working in this tenant.
→ iManage · SharePoint · Email · Network
p95 — Individual History
95th percentile of each user's personal 30-day history. Right for stable, predictable behaviour that doesn't spike legitimately. Simpler to understand, easier to explain to stakeholders. Correct model when behaviour is consistent enough that a p95 threshold remains meaningful over time.
→ File Access · Logon Patterns · Print Volume
Presence Detection — Array Lookup
Not a counting problem. The first occurrence of an activity is the signal — volume is secondary. Builds an array of known-good items per user. Detection rules check whether a new event is in or out of that array. No statistics required. Right for USB devices, process execution scope, and system access scope.
→ USB Devices · System Access · Process Scope

// ITM Baseline — iManage Document Access MAD Cohort
// Output → baseline_imanage_mad
// Model: MAD cohort by jobTitle peer group
// Dataset: imanage_threat_manager_generic_alert_raw  (Threat Manager alerts NOT raw events)
// Used by: T1.1 Bulk Data Access, T1.2 Sensitive Repository Enumeration
// Schedule: Daily 00:30 | Run Now immediately after creating
// Field names confirmed from tenant testing

config timeframe = 30d

| dataset = imanage_threat_manager_generic_alert_raw
| filter _alert_data != null
| filter not (_raw ~= "(?i)anomalous")

// Extract counts from _raw JSON — confirmed field paths
| alter doc_count    = to_integer(json_extract_scalar(_raw, "$.tm_counts['document count']"))
| alter matter_count = to_integer(json_extract_scalar(_raw, "$.tm_counts['matter count']"))
| alter client_count = to_integer(json_extract_scalar(_raw, "$.tm_counts['client count']"))

// Identity
| alter uid        = lowercase(user_email)
| alter NowEpoch   = to_epoch(current_time())
| alter EventEpoch = to_epoch(_time)
| alter DaysAgo    = to_integer(divide(subtract(NowEpoch, EventEpoch), 86400000))

| filter uid != null and uid != ""
| filter not (uid ~= "^(sa-|svc_|service_|iman|wsadmin|system)")

// Enrich with jobTitle
| join type = left (
    dataset = msft_o365_users_raw
    | alter J = lowercase(mail)
    | fields J, jobTitle, department
  ) as u u.J = uid

// Pass 1 — per-user 7-day totals
| comp
    avg(doc_count)                         as user_mean_30d,
    sum(if(DaysAgo <= 7, doc_count, 0))    as recent_total_7d,
    sum(if(DaysAgo <= 7, matter_count, 0)) as recent_matters_7d,
    sum(if(DaysAgo <= 7, client_count, 0)) as recent_clients_7d,
    max(jobTitle)                          as jobTitle,
    max(department)                        as department,
    max(user_email)                        as user_email
  by uid

| alter recent_avg_7d = divide(recent_total_7d, 7)
| filter recent_total_7d > 0
| filter jobTitle != null and jobTitle != ""

// Pass 2 — cohort stats (Mean + MAD) by jobTitle
| join type = left (
    dataset = imanage_threat_manager_generic_alert_raw
    | filter _alert_data != null
    | filter not (_raw ~= "(?i)anomalous")
    | alter doc_count  = to_integer(json_extract_scalar(_raw, "$.tm_counts['document count']"))
    | alter uid        = lowercase(user_email)
    | alter NowEpoch   = to_epoch(current_time())
    | alter EventEpoch = to_epoch(_time)
    | alter DaysAgo    = to_integer(divide(subtract(NowEpoch, EventEpoch), 86400000))
    | filter uid != null and uid != ""
    | filter not (uid ~= "^(sa-|svc_|service_|iman|wsadmin|system)")
    | join type = left (
        dataset = msft_o365_users_raw
        | alter J = lowercase(mail)
        | fields J, jobTitle
      ) as u2 u2.J = uid
    | filter jobTitle != null and jobTitle != ""
    | comp sum(if(DaysAgo <= 7, doc_count, 0)) as recent_total_7d
      by uid, jobTitle
    | alter recent_avg_7d = divide(recent_total_7d, 7)
    | filter recent_total_7d > 0
    | join type = left (
        dataset = imanage_threat_manager_generic_alert_raw
        | filter _alert_data != null
        | filter not (_raw ~= "(?i)anomalous")
        | alter doc_count  = to_integer(json_extract_scalar(_raw, "$.tm_counts['document count']"))
        | alter uid        = lowercase(user_email)
        | alter NowEpoch   = to_epoch(current_time())
        | alter EventEpoch = to_epoch(_time)
        | alter DaysAgo    = to_integer(divide(subtract(NowEpoch, EventEpoch), 86400000))
        | filter uid != null and uid != ""
        | filter not (uid ~= "^(sa-|svc_|service_|iman|wsadmin|system)")
        | join type = left (
            dataset = msft_o365_users_raw
            | alter J = lowercase(mail)
            | fields J, jobTitle
          ) as u3 u3.J = uid
        | filter jobTitle != null and jobTitle != ""
        | comp sum(if(DaysAgo <= 7, doc_count, 0)) as recent_total_7d
          by uid, jobTitle
        | alter recent_avg_7d = divide(recent_total_7d, 7)
        | filter recent_total_7d > 0
        | comp
            avg(recent_avg_7d) as cohort_mean_recent,
            count()            as cohort_users
          by jobTitle
        | filter cohort_users >= 3
        | fields jobTitle,
                 cohort_mean_recent as mean_m,
                 cohort_users       as users_m
      ) as m m.jobTitle = jobTitle
    | alter abs_dev = if(
        recent_avg_7d >= mean_m,
        subtract(recent_avg_7d, mean_m),
        subtract(mean_m, recent_avg_7d)
      )
    | comp
        max(mean_m)  as cohort_mean_recent,
        max(users_m) as cohort_users,
        avg(abs_dev) as cohort_mad_recent
      by jobTitle
    | fields jobTitle, cohort_mean_recent, cohort_mad_recent, cohort_users
  ) as coh coh.jobTitle = jobTitle

// Adaptive threshold
| filter cohort_mean_recent != null and cohort_mad_recent != null
| alter safe_mad    = if(cohort_mad_recent > 0, cohort_mad_recent, 1)
| alter k_mult      = 2.0
| alter upper_bound = add(cohort_mean_recent, multiply(k_mult, safe_mad))
| alter last_updated = current_time()

| fields uid as actor_effective_username,
         user_email, jobTitle, department,
         recent_avg_7d,
         recent_matters_7d, recent_clients_7d,
         user_mean_30d,
         cohort_mean_recent, cohort_mad_recent,
         cohort_users, upper_bound, k_mult,
         last_updated

// ITM Baseline — External Email MAD Cohort
// Output → baseline_email_mad
// Used by: T3.1 External Email Sharing
// *** Replace aoshearman.com with your domain ***

dataset = email_data
| filter message_directionality = "outgoing"

| alter sender_address = sender -> address
| alter to_text = to_string(recipients)
| alter recipient_domain = extract(to_text, "@([^@\"]+)")
| filter recipient_domain != "aoshearman.com"
    and recipient_domain != null

// Join jobTitle for cohort grouping
| join type = left (
    dataset = msft_o365_users_raw
    | fields userPrincipalName, jobTitle
    | dedup userPrincipalName by asc _time
  ) as u
  u.userPrincipalName = mailbox_owner

| alter is_recent = if(
    _time > to_epoch(now() - duration(7d)), 1, 0
  )

// Pass 1: Per user
| comp
    count() as total_30d,
    sum(is_recent) as total_7d,
    values(recipient_domain) as known_domains
  by mailbox_owner, u.jobTitle as job_title

| alter avg_7d  = divide(to_float(total_7d), 7.0)

// Pass 2: Cohort mean
| join type = left (
    dataset = email_data
    | filter message_directionality = "outgoing"
    | alter sender_address = sender -> address
    | alter to_text = to_string(recipients)
    | alter rd = extract(to_text, "@([^@\"]+)")
    | filter rd != "aoshearman.com" and rd != null
    | join type = left (
        dataset = msft_o365_users_raw
        | fields userPrincipalName, jobTitle
        | dedup userPrincipalName by asc _time
      ) as u2 u2.userPrincipalName = mailbox_owner
    | alter ir = if(_time > to_epoch(now() - duration(7d)),1,0)
    | comp sum(ir) as u7d
      by mailbox_owner, u2.jobTitle as jt
    | alter ua = divide(to_float(u7d), 7.0)
    | comp avg(ua) as cohort_mean by jt as ct
  ) as cohort cohort.ct = job_title

// Pass 3: MAD
| alter abs_dev = if(
    avg_7d >= cohort.cohort_mean,
    subtract(avg_7d, cohort.cohort_mean),
    subtract(cohort.cohort_mean, avg_7d)
  )
| alter safe_mad = if(abs_dev > 0, abs_dev, 1.0)
| alter upper_bound = add(
    cohort.cohort_mean,
    multiply(safe_mad, 2.0)
  )

| fields
    mailbox_owner as actor_effective_username,
    job_title,
    avg_7d,
    cohort.cohort_mean as cohort_mean,
    upper_bound,
    known_domains

// ITM Baseline — Network Egress MAD Cohort
// Output → baseline_network_mad
// Used by: T3.4 Network Exfiltration, T8.1 DoS

dataset = panw_ngfw_traffic_raw
| filter source_user != null
    and source_user != ""
    and action = "allow"

// Hourly aggregation per user
| bin _time span = 1h
| comp
    sum(bytes_sent) as hourly_bytes_out,
    sum(packets_sent) as hourly_packets,
    count() as hourly_sessions
  by source_user, _time

// Join jobTitle for cohort grouping
| join type = left (
    dataset = msft_o365_users_raw
    | fields userPrincipalName, jobTitle
    | dedup userPrincipalName by asc _time
  ) as u
  u.userPrincipalName = source_user

// Per-user hourly stats
| comp
    avg(hourly_bytes_out) as avg_hourly_bytes,
    avg(hourly_packets) as avg_hourly_packets,
    count() as hours_observed,
    max(_time) as last_seen
  by source_user, u.jobTitle as job_title

// Cohort mean per job title
| join type = left (
    dataset = panw_ngfw_traffic_raw
    | filter source_user != null and action = "allow"
    | bin _time span = 1h
    | comp sum(bytes_sent) as hb, sum(packets_sent) as hp
      by source_user, _time
    | join type = left (
        dataset = msft_o365_users_raw
        | fields userPrincipalName, jobTitle
        | dedup userPrincipalName by asc _time
      ) as u2 u2.userPrincipalName = source_user
    | comp avg(hb) as u_avg_b, avg(hp) as u_avg_p
      by source_user, u2.jobTitle as jt
    | comp
        avg(u_avg_b) as cohort_mean_bytes,
        avg(u_avg_p) as cohort_mean_packets
      by jt as ct
  ) as cohort cohort.ct = job_title

// MAD for bytes
| alter abs_dev_bytes = if(
    avg_hourly_bytes >= cohort.cohort_mean_bytes,
    subtract(avg_hourly_bytes, cohort.cohort_mean_bytes),
    subtract(cohort.cohort_mean_bytes, avg_hourly_bytes)
  )
| alter safe_mad = if(abs_dev_bytes > 0, abs_dev_bytes, 1.0)
| alter upper_bound_bytes = add(
    cohort.cohort_mean_bytes,
    multiply(safe_mad, 2.0)
  )

| fields
    source_user as actor_effective_username,
    job_title,
    avg_hourly_bytes,
    avg_hourly_packets,
    cohort.cohort_mean_bytes as cohort_mean_bytes,
    cohort.cohort_mean_packets as cohort_mean_packets,
    upper_bound_bytes,
    hours_observed,
    last_seen

Presence Detection — Array Baselines
Not a counting problem. For USB, system access, and process execution — the first occurrence is already the signal. These baselines build arrays of known-good items. Detection rules check whether new events are inside or outside that array. A user with a null row in the USB table has never used USB — any USB event is immediately high confidence regardless of volume.

// ITM Baseline — Process Execution Scope
// Output → baseline_processes
// Model: Presence array — which processes has this user ever launched
// Used by: T2.2 Workflow Deviation
// Schedule: Weekly Sunday 02:30 | 60-day lookback

config timeframe = 60d

dataset = xdr_data
| filter event_type = ENUM.PROCESS
    and event_sub_type = ENUM.PROCESS_START
    and actor_effective_username != null
| filter not (actor_effective_username contains "NT AUTHORITY")
| filter not (actor_effective_username contains "SYSTEM")
| filter not (actor_effective_username contains "$")
| filter actor_process_image_name != null
| filter not (actor_process_image_name in (
    "svchost.exe", "lsass.exe", "csrss.exe",
    "wininit.exe", "winlogon.exe", "services.exe",
    "smss.exe", "System", "Registry"
  ))
| comp
    values(actor_process_image_name)         as known_processes,
    values(actor_process_image_sha256)       as known_hashes,

    max(_time)                               as last_updated
  by actor_effective_username
| filter total_executions >= 5
| fields actor_effective_username,
         known_processes, known_hashes,
         unique_process_count,
         total_executions, last_updated
